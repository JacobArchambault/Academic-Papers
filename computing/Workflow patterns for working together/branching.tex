	\section{Current challenges and their canonical solutions}
	\begin{frame}{Challenges}
		\begin{itemize}
			\item coordinating multiple teams working in the same repository \pause
			\item lack of direct collaboration, and occasionally trust, between development, operations, and testing \pause
			\item late integration of contributor code \pause
			\begin{itemize}
				\item feedback delays \pause
				\item merge conflict regressions \pause
				\item sunk costs fallacy risks \pause
			\end{itemize}
			\item coupling of all services at the database layer \pause
			\begin{itemize}
				\item high storage costs \pause
				\item single point of failure risk \pause
				\item risk of cross-project interference \pause
			\end{itemize}
		\end{itemize}
	\end{frame}
	\begin{frame}{Canonical solutions to these challenges}
		\begin{itemize}
			\item multiple teams $\rightarrow$ inverse Conway maneuver \pause
			\item collaboration $\rightarrow$ business stream-aligned teams over activity oriented teams \pause
			\item late integration $\rightarrow$ adopting a true continuous integration workflow \pause
			\item database coupling $\rightarrow$ Bezos API mandate/``Microservices'' \pause
			
			
		\end{itemize}
	\end{frame}

	\subsection{Communication}
	\subsubsection{Conway's Law}
	\begin{frame}{1967: Conway's law}
		\begin{quote}
			`Organizations which design systems [...] are constrained to produce designs which are copies of the communication structures of these organizations.' - Melvin Conway, `How do Committees Invent?' Datamation, 1967
		\end{quote}
	\end{frame}
	\begin{frame}{Conway's law: examples}
		\begin{itemize}
			\item `If you have four groups working on a compiler, you'll get a 4-pass compiler' - The New Hacker's Dictionary, 1996\pause
			\item front-end [devs] business layer [backend devs] monolithic database [DBA team] \pause
			\item a web api controller [manager]  delegates most business logic to business classes [developers] which are part of the same in-memory process [team], while serving as a single entry-point for wider cross-network [cross-team] communication.
		\end{itemize}
	\end{frame}
	\begin{frame}{Conway's law: more examples}
		\begin{itemize}
			\item separate \texttt{.csproj} projects for tests written by QA \pause
			\item separate development and config repositories for the same transaction signifying low communication between development and operations \pause
			\item all business documentation in the same repository, separate from the repositories that the documentation is for \pause
		\end{itemize}
	\end{frame}
	\begin{frame}{Conway's law: corollaries}
		\begin{itemize}
			\item Conway's law can't be fought \pause
			\item Teams working in the same part of the same codebase at the same time are the same team \pause
			\begin{itemize}
				\item separate the codebases for improved velocity \pause
				\item join the teams for improved code sharing and communication (Inverse conway maneuver) \pause 
				\item branching-based strategies for team management compromise between the above approaches without their benefits
			\end{itemize}
		\end{itemize}
	\end{frame}

	\section{Communication and throughput anti-patterns and solutions}
	\subsection{Communication anti-pattern: local optimization}
	\begin{frame}{communication anti-pattern: local optimization}
		\begin{itemize}
			\item example: separate (DevOps, QA, operations, support) team \pause
			\item leads to bottlenecks \pause
			\item delays communication \pause
			\item discourages cross-functional collaboration \pause
			\item punishes untracked work (e.g. thorough code reviews/dev testing, helping blocked colleagues)
		\end{itemize}
	\end{frame}
	\begin{frame}{Solution}
		\begin{itemize}
			\item autonomous, cross-functional teams with `t-shaped' employees \pause
			\item communities of practice \pause
			\item A \textit{team} is the minimal socio-technical unit that can deliver a production change with no additional coordination beyond mandated governance or custodial controls. \pause
			\begin{itemize}
				\item pull requests \pause
				\item testing \pause
				\item deployment \pause
			\end{itemize}
		\end{itemize}
	\end{frame}
