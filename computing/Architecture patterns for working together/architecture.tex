	\section{Conway's law}
	\begin{frame}{1967: Conway's law}
		\begin{quote}
			`Organizations which design systems [...] are constrained to produce designs which are copies of the communication structures of these organizations.' - Melvin Conway, `How do Committees Invent?' Datamation, 1967
		\end{quote}
	\end{frame}
	\begin{frame}{Conway's law: examples}
		\begin{itemize}
			\item separate \texttt{.csproj} projects for tests written by QA 
			\item separate development and config repositories for the same transaction signifying low communication between development and operations 
			\item all business documentation in the same repository, separate from the repositories that the documentation is for 
		\end{itemize}
	\end{frame}
	\begin{frame}{Conway's law: corollaries}
		\begin{itemize}
			\item Conway's law can't be fought \pause
			\item Teams working in the same part of the same codebase at the same time are the same team \pause
			\begin{itemize}
				\item separate the codebases for improved velocity \pause
				\item join the teams for improved code sharing and communication (Inverse conway maneuver) \pause 
				\item branching-based strategies for team management compromise between the above approaches without their benefits
			\end{itemize}
		\end{itemize}
	\end{frame}
	\begin{frame}{Revisiting Conway's law: defining a team}
		\begin{itemize}
			\item A \textit{team} is the minimal socio-technical unit that can deliver a production change with no additional coordination beyond mandated governance or custodial controls. \pause
			\begin{itemize}
				\item pull requests \pause
				\item testing \pause
				\item deployment \pause
			\end{itemize}
		\end{itemize}
	\end{frame}

	\subsection{Theory}
	\begin{frame}{A naive probability theory for merge consistency: isolated commits}
		\begin{itemize}
			\item $C = \{a, b, c\}$ \pause
			\item $\mathcal{P}(C) = \{ \{\emptyset\}, \{a\}, \{b\}, \{c\}, \{a, b\}, \{a, c\}, \{b, c\}, \{a, b, c\}\}$ \pause
			\item $P$(\textit{no conflicts}) = $\mathcal{P}(\emptyset)$ = $\frac{1}{8}$ = $12.5\%$ \pause
			\item $P$(\textit{no conflicts}) = $\frac{1}{2^n}$ where $|C| = n$ \pause
			\begin{itemize}
				\item $P$(\textit{no conflicts}) $< 10\%$ where $n \geq 4$ \pause
				\item $P$(\textit{no conflicts})  $< 1\%$ where $n \geq 7$ \pause
			\end{itemize}
		\end{itemize}
	\end{frame}
	\begin{frame}{A naive probability theory for merge consistency: modeling branches as sets of commits}
		\begin{itemize}
			\item $A = A_1 \sqcup A_2 \sqcup A_3$ where $A_1 = \{a\}$, $A_2 = \{b, c\}$, $A_3 = \{d, e\}$ \pause
			\item Assume $A_1 \cdots A_3$ are known good sets of commits. Then $P$(\textit{no conflicts}($A$)) = probability of selecting the null set from the set containing both the null set and all remaining unchecked potential conflict sets  \pause
			\begin{itemize}
				\item $= \frac{1}{2^n - (2^{n_1} - 1) - (2^{n_2} - 1) - (2^{n_3} - 1)}$ where $n = |A|$ and $n_i = |A_i|$ \pause
				\item $= \frac{1}{2^n - 2^{n_1} - 2^{n_2} - 2^{n_3} + 3}$ \pause
				\item $= \frac{1}{2^5 - 2^1 - 2^2 - 2^2 + 3}$ \pause 
				\item $ = \frac{1}{32 - 2 - 4 - 4 + 3}$ \pause 
				\item $= \frac{1}{32 - 7}$ \pause
				\item $= \frac{1}{25}$ \pause
				\item $= 4\%$ \pause		
			\end{itemize}
		\end{itemize}
	\end{frame}
	\begin{frame}{A naive probability theory for merge consistency: general definition for distinct branches diverging from a common ancestor}
		\begin{itemize}
			\item Let $A$ be the set of unmerged commits across $k$ distinct branches $A_1 \cdots A_k$, where $A = A_1 \sqcup A_2 \sqcup \cdots \sqcup A_k$, $n = |A|$, and $n_i = |A_i|$ for $1 \leq i \leq k$. Then \pause
		\end{itemize}
		\[
		P(\textit{no conflicts}) = \frac{1}{2^n - (\sum_{i=1}^k2^{n_i}) + k}
		\]
	\end{frame}

	\section{Going forward}
	\subsection{Team composition: stream aligned teams}
	\begin{frame}{Going forward: team composition}
		\begin{itemize}
			\item All teams working in same git repository are in the same daily scrum \pause
			\item Each team in the above sense has its own Microsoft Teams chat with only its own contributors in it. I will set this up. \pause
			\item Teams are \textit{stream-aligned}, i.e., they focus on a particular business area (in our case, either a transaction type or a closely related set of transaction types) \pause
			\item Teams can move work forward without waiting on or requiring outside assistance from another team \pause	
		\end{itemize}
	\end{frame}
	\subsection{Branching and committing}
	\begin{frame}{Going forward: branching and committing}
		\begin{itemize}
			\item All commit work done by a given team, by both developers and testers, is committed directly to the same shared branch \pause
			\begin{itemize}
				\item Note: resist the temptation to branch off the shared branch \pause
			\end{itemize}
			\item Every commit is pushed to origin only and immediately after passing local testing \pause
			\item Every git commit message begins with a Jira ticket number \pause
		\end{itemize}
	\end{frame}
	\subsection{Testing}
	\begin{frame}{Going forward: testing}
		\begin{itemize}
			\item Every change is tested by the test suite locally before committing \pause
			\item Tests are written against new code as that code is being written  \pause
			\item We don't push up non-passing tests. Instead, we find the person we need to talk to and work out a passing solution, then push up. \pause
			\item Testing need not be end-to-end, but may simply test a component \pause
			\item Goal is to move most testing into the test suite so that manual checks become minimal. \pause
		\end{itemize}
	\end{frame}
	\subsection{Pull requests}
	\begin{frame}{Going forward: pull requests}
		\begin{itemize}
			\item Pull requests occur at least daily \pause
			\item Because the working branch is by hypothesis always releasable, any contributor can PR the working branch at any time in principle. In practice, this will be worked out informally beforehand. \pause
			\item To allow time for review before end of day, pull requests should generally be set up by 3-3:30pm eastern time. \pause
			\item Because leads are also both contributing to and seeing changes to the working branch in real time, in practice PR's becomes increasingly \textit{pro forma} \pause
		\end{itemize}
	\end{frame}
	\begin{frame}{Going forward: branching strategies for frequent pull requests}
		\begin{itemize}
			\item Continuing branch, e.g., PEDI-XXXX-feature\_description \pause
			\begin{itemize}
				\item Branch created, PR created, branch merged, develop merged into PR branch, work continues until feature is complete \pause
			\end{itemize}
			\item Stacked daily branches, e.g., PEDI-XXXX-feature\_description-YYYYMMDD \pause
			\begin{itemize}
				\item Branch created, PR created, new branch created off of PR branch, work continues, backmerge develop as needed \pause
			\end{itemize}
		\end{itemize}
	\end{frame}
