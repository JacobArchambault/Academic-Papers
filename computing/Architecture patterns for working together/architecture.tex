\section{Current challenges and their canonical solutions}
\begin{frame}{Challenges}
	\begin{itemize}
		\item multiple teams working in the same repository \pause
		\item lack of direct collaboration, and occasionally trust, between development, operations, and testing \pause
		\item late integration of contributor code \pause
		\begin{itemize}
			\item feedback delays \pause
			\item merge conflict regressions \pause
			\item sunk costs fallacy risks \pause
		\end{itemize}
		\item coupling of all services at the database layer \pause
		\begin{itemize}
			\item high storage costs \pause
			\item single point of failure risk \pause
			\item risk of cross-project interference \pause
		\end{itemize}
	\end{itemize}
\end{frame}
\begin{frame}{Canonical solutions to these challenges}
	\begin{itemize}
		\item multiple teams $\rightarrow$ inverse Conway maneuver \pause
		\item collaboration $\rightarrow$ business stream-aligned teams over activity oriented teams \pause
		\item late integration $\rightarrow$ adopting a true continuous integration workflow \pause
		\item database coupling $\rightarrow$ Bezos API mandate/``Microservices'' \pause		
	\end{itemize}
\end{frame}
\section{Going forward}
\subsection{Team Composition}
\begin{frame}{Team composition}
\begin{itemize}
	\item all teams in same repo are in the same daily scrum \pause
	\item each team in the above sense has its own small Microsoft Teams channel with only its own contributors in it \pause
	\item these teams are \textit{stream-aligned}, i.e., they focus on a particular business area (in our case, a transaction type) \pause
	\item these teams are \textit{autonomous}, i.e., they can move work forward without requiring outside assistance from another team \pause	
\end{itemize}
\end{frame}
