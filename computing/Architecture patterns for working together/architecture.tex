\section{Current challenges and their canonical solutions}
\begin{frame}{Challenges}
	\begin{itemize}
		\item multiple teams working in the same repository \pause
		\item lack of direct collaboration, and occasionally trust, between development, operations, and testing \pause
		\item late integration of contributor code \pause
		\begin{itemize}
			\item feedback delays \pause
			\item merge conflict regressions \pause
			\item sunk costs fallacy risks \pause
		\end{itemize}
		\item coupling of all services at the database layer \pause
		\begin{itemize}
			\item high storage costs \pause
			\item single point of failure risk \pause
			\item risk of cross-project interference \pause
		\end{itemize}
	\end{itemize}
\end{frame}
\begin{frame}{Canonical solutions to these challenges}
	\begin{itemize}
		\item multiple teams $\rightarrow$ inverse Conway maneuver \pause
		\item collaboration $\rightarrow$ business stream-aligned teams over activity oriented teams \pause
		\item late integration $\rightarrow$ adopting a true continuous integration workflow \pause
		\item database coupling $\rightarrow$ Bezos API mandate/``Microservices'' \pause		
	\end{itemize}
\end{frame}
\section{Branching: theory and statistics}
\subsection{Theory}
\begin{frame}{A naive probability theory for merge consistency: isolated commits}
	\begin{itemize}
		\item $C = \{a, b, c\}$ \pause
		\item $\mathcal{P}(C) = \{ \{\emptyset\}, \{a\}, \{b\}, \{c\}, \{a, b\}, \{a, c\}, \{b, c\}, \{a, b, c\}\}$ \pause
		\item $P$(\textit{no conflicts}) = $\mathcal{P}(\emptyset)$ = $\frac{1}{8}$ = $12.5\%$ \pause
		\item $P$(\textit{no conflicts}) = $\frac{1}{2^n}$ where $|C| = n$ \pause
		\begin{itemize}
			\item $P$(\textit{no conflicts}) $< 10\%$ where $n \geq 4$ \pause
			\item $P$(\textit{no conflicts})  $< 1\%$ where $n \geq 7$ \pause
		\end{itemize}
	\end{itemize}
\end{frame}
\begin{frame}{A naive probability theory for merge consistency: modeling branches as sets of commits}
	\begin{itemize}
		\item $A = A_1 \sqcup A_2 \sqcup A_3$ where $A_1 = \{a\}$, $A_2 = \{b, c\}$, $A_3 = \{d, e\}$ \pause
		\item Assume $A_1 \cdots A_3$ are known good sets of commits. Then $P$(\textit{no conflicts}($A$)) = probability of selecting the null set from the set containing both the null set and all remaining unchecked potential conflict sets  \pause
		\begin{itemize}
			\item $= \frac{1}{2^n - (2^{n_1} - 1) - (2^{n_2} - 1) - (2^{n_3} - 1)}$ where $n = |A|$ and $n_i = |A_i|$ \pause
			\item $= \frac{1}{2^n - 2^{n_1} - 2^{n_2} - 2^{n_3} + 3}$ \pause
			\item $= \frac{1}{2^5 - 2^1 - 2^2 - 2^2 + 3}$ \pause 
			\item $ = \frac{1}{32 - 2 - 4 - 4 + 3}$ \pause 
			\item $= \frac{1}{32 - 7}$ \pause
			\item $= \frac{1}{25}$ \pause
			\item $= 4\%$ \pause		
		\end{itemize}
	\end{itemize}
\end{frame}
\begin{frame}{A naive probability theory for merge consistency: general definition for distinct branches diverging from a common ancestor}
	\begin{itemize}
		\item Let $A$ be the set of unmerged commits across $k$ distinct branches $A_1 \cdots A_k$, where $A = A_1 \sqcup A_2 \sqcup \cdots \sqcup A_k$, $n = |A|$, and $n_i = |A_i|$ for $1 \leq i \leq k$. Then \pause
	\end{itemize}
	\[
	P(\textit{no conflicts}) = \frac{1}{2^n - (\sum_{i=1}^k2^{n_i}) + k}
	\]
\end{frame}

\section{Going forward}
\subsection{Team Composition}
\begin{frame}{Team composition}
\begin{itemize}
	\item all teams working in same git repository are in the same daily scrum \pause
	\item each team in the above sense has its own Microsoft Teams chat with only its own contributors in it \pause
	\item these teams are \textit{stream-aligned}, i.e., they focus on a particular business area (in our case, either a transaction type or a closely related set of transaction types) \pause
	\item these teams are \textit{autonomous}, i.e., they can move work forward without requiring outside assistance from another team \pause	
\end{itemize}
\end{frame}
\begin{frame}{Team composition: enabling teams}
	\begin{itemize}
		\item Strategy will be rolled out first for claims, then reimbursements, then other transactions \pause
		\item the 837 COB team will serve as an enabling team for the new branching strategy \pause
	\end{itemize}
\end{frame}
\subsection{Branching and committing}
\begin{frame}{Branching and committing}
	\begin{itemize}
		\item all commit work done by a given team, by both developers and testers, is committed directly to the same shared branch \pause
		\item every commit is pushed  to origin after passing local testing \pause
		\item every git commit message begins with a Jira ticket number \pause
	\end{itemize}
\end{frame}
\subsection{Testing}
\begin{itemize}
	\item Every change is tested by the test suite locally before committing \pause
	\item tests are written against new code as that code is being written \pause
\end{itemize}
